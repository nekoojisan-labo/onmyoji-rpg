  ```typescript
  // module: engine/loop | onmyoji-rpg | 固定タイムステップの更新ループ（requestAnimationFrame ラッパ）
  // 契約 §4.19 逐語: FIXED_DT / class GameLoop(update, render).start()/.stop()

  /** 固定タイムステップ秒（契約 §4.19）。論理更新は常にこの刻みで進む */
  export const FIXED_DT = 1 / 60;

  /** 1フレームで消化する最大ステップ数（spiral of death 防止）。タブ復帰等の巨大Δtをクランプ */
  const MAX_STEPS_PER_FRAME = 5;

  export class GameLoop {
    private readonly update: (dt: number) => void;
    private readonly render: (alpha: number) => void;
    private accumulator = 0;
    private running = false;
    private rafId = 0;
    private lastMs = 0;

    constructor(update: (dt: number) => void, render: (alpha: number) => void) {
      this.update = update;
      this.render = render;
    }

    /**
     * 経過秒を蓄積し、固定ステップ単位で update を呼び、残差を補間係数 alpha にして render を呼ぶ。
     * raf から呼ばれる本体。テストからは elapsed を直接与えて決定的に検証する。
     */
    stepForElapsed(elapsedSec: number): void {
      this.accumulator += elapsedSec;
      const maxAccumulated = FIXED_DT * MAX_STEPS_PER_FRAME;
      if (this.accumulator > maxAccumulated) {
        this.accumulator = maxAccumulated;
      }
      while (this.accumulator >= FIXED_DT) {
        this.update(FIXED_DT);
        this.accumulator -= FIXED_DT;
      }
      const alpha = this.accumulator / FIXED_DT;
      this.render(alpha);
    }

    start(): void {
      if (this.running) return;
      this.running = true;
      this.lastMs = performance.now();
      const frame = (nowMs: number) => {
        if (!this.running) return;
        const elapsedSec = (nowMs - this.lastMs) / 1000;
        this.lastMs = nowMs;
        this.stepForElapsed(elapsedSec);
        this.rafId = requestAnimationFrame(frame);
      };
      this.rafId = requestAnimationFrame(frame);
    }

    stop(): void {
      this.running = false;
      if (this.rafId !== 0) {
        cancelAnimationFrame(this.rafId);
        this.rafId = 0;
      }
    }
  }
  ```

  ```typescript
  // module: engine/camera | onmyoji-rpg | OrthographicCamera を yaw45/pitch30 固定生成・追従pan・ピクセルスナップ・境界クランプ
  // 契約 §4.9 逐語: class IsoCamera(viewWorldWidth, aspect) / follow / clampToBounds / applyPixelSnap / resize

  import * as THREE from 'three';
  import type { Rect } from '../types';
  import {
    CAMERA_YAW_DEG,
    CAMERA_PITCH_DEG,
    CAMERA_NEAR,
    CAMERA_FAR,
  } from '../config';

  /** カメラをターゲットから引いて置く距離（オルソなので見た目サイズには無関係。フラスタム内に収める） */
  const CAMERA_DISTANCE = 100;

  /**
   * クラシックアイソメの固定カメラ。回転せず、ターゲットを画面中央に保つよう平行移動のみ行う。
   * 向き = yaw45（水平回り）/ pitch30（見下ろし）。
   */
  export class IsoCamera {
    readonly camera: THREE.OrthographicCamera;
    private viewWorldWidth: number;
    /** ターゲット注視点（XZ平面・float）。follow が更新し、clamp が補正する */
    private targetX = 0;
    private targetZ = 0;
    /** カメラ→ターゲットの単位方向（yaw45/pitch30 から一度だけ算出して固定） */
    private readonly forward: THREE.Vector3;

    constructor(viewWorldWidth: number, aspect: number) {
      this.viewWorldWidth = viewWorldWidth;
      const yaw = THREE.MathUtils.degToRad(CAMERA_YAW_DEG);
      const pitch = THREE.MathUtils.degToRad(CAMERA_PITCH_DEG);
      // 見下ろし前方ベクトル: 水平成分長 cos(pitch)・下向き -sin(pitch)・水平は yaw 回り
      this.forward = new THREE.Vector3(
        Math.cos(pitch) * Math.sin(yaw),
        -Math.sin(pitch),
        Math.cos(pitch) * Math.cos(yaw),
      ).normalize();

      const halfW = viewWorldWidth / 2;
      const halfH = halfW / aspect;
      this.camera = new THREE.OrthographicCamera(
        -halfW, halfW, halfH, -halfH, CAMERA_NEAR, CAMERA_FAR,
      );
      this.applyPlacement();
    }

    /** targetX/targetZ を中央に保つようカメラ位置と向きを更新する（回転は forward 固定なので変わらない） */
    private applyPlacement(): void {
      const target = new THREE.Vector3(this.targetX, 0, this.targetZ);
      // カメラはターゲットから forward の逆方向へ CAMERA_DISTANCE 引いた位置に置く
      this.camera.position.copy(target).addScaledVector(this.forward, -CAMERA_DISTANCE);
      this.camera.up.set(0, 1, 0);
      this.camera.lookAt(target);
      this.camera.updateMatrixWorld(true);
    }

    /** デッドゾーン無しの即時追従（デッドゾーン付きlerpは Task G2.4 / G4 で詰める） */
    follow(targetX: number, targetZ: number): void {
      this.targetX = targetX;
      this.targetZ = targetZ;
      this.applyPlacement();
    }

    /** 注視点をマップ境界(Rect)内にクランプする */
    clampToBounds(bounds: Rect): void {
      const minX = bounds.x - bounds.w / 2;
      const maxX = bounds.x + bounds.w / 2;
      const minZ = bounds.z - bounds.h / 2;
      const maxZ = bounds.z + bounds.h / 2;
      this.targetX = THREE.MathUtils.clamp(this.targetX, minX, maxX);
      this.targetZ = THREE.MathUtils.clamp(this.targetZ, minZ, maxZ);
      this.applyPlacement();
    }

    /** 滲み防止: 注視点を内部解像度ピクセルグリッドへスナップする（ワールド単位/ピクセル換算で丸め） */
    applyPixelSnap(internalWidth: number, internalHeight: number): void {
      void internalHeight;
      const worldPerPixel = this.viewWorldWidth / internalWidth;
      const snappedX = Math.round(this.targetX / worldPerPixel) * worldPerPixel;
      const snappedZ = Math.round(this.targetZ / worldPerPixel) * worldPerPixel;
      const target = new THREE.Vector3(snappedX, 0, snappedZ);
      this.camera.position.copy(target).addScaledVector(this.forward, -CAMERA_DISTANCE);
      this.camera.lookAt(target);
      this.camera.updateMatrixWorld(true);
    }

    /** アスペクト変更時にフラスタム高さを再計算（幅= viewWorldWidth 固定） */
    resize(aspect: number): void {
      const halfW = this.viewWorldWidth / 2;
      const halfH = halfW / aspect;
      this.camera.left = -halfW;
      this.camera.right = halfW;
      this.camera.top = halfH;
      this.camera.bottom = -halfH;
      this.camera.updateProjectionMatrix();
    }
  }
  ```

  ```typescript
  // module: engine/renderer | onmyoji-rpg | WebGLRenderer/Scene/グループ初期化・環境光・藍墨fog・resize追従
  // 契約 §3 file structure / §2 内部解像度1280×720・fog #10131c near12 far40

  import * as THREE from 'three';
  import {
    INTERNAL_WIDTH,
    INTERNAL_HEIGHT,
    FOG_COLOR,
    FOG_NEAR,
    FOG_FAR,
    AMBIENT_COLOR,
    AMBIENT_INTENSITY,
  } from '../config';

  export class Renderer {
    readonly renderer: THREE.WebGLRenderer;
    readonly scene: THREE.Scene;
    readonly worldGroup: THREE.Group;
    readonly entityGroup: THREE.Group;
    readonly fxGroup: THREE.Group;
    readonly shadowGroup: THREE.Group;

    constructor(canvas: HTMLCanvasElement) {
      this.renderer = new THREE.WebGLRenderer({
        canvas,
        antialias: false,         // ピクセルアート滲み防止のためAA無効
        powerPreference: 'high-performance',
      });
      // 内部解像度を固定（整数ピクセル）。CSSサイズは resize で等比フィットさせる。
      this.renderer.setPixelRatio(1);
      this.renderer.setSize(INTERNAL_WIDTH, INTERNAL_HEIGHT, false);
      this.renderer.setClearColor(FOG_COLOR, 1);
      // ニアレスト相当の見え方を保つためトーンマッピングは線形（過度な減衰を避ける）
      this.renderer.outputColorSpace = THREE.SRGBColorSpace;

      this.scene = new THREE.Scene();
      this.scene.background = new THREE.Color(FOG_COLOR);
      // 藍墨フォグ（near=12u / far=40u）。和風ダークの空気遠近。
      this.scene.fog = new THREE.Fog(FOG_COLOR, FOG_NEAR, FOG_FAR);

      // 和風ダーク用の最小ライティング = 寒色の環境光のみ（ビルボード前提で平面照明）。
      const ambient = new THREE.AmbientLight(AMBIENT_COLOR, AMBIENT_INTENSITY);
      this.scene.add(ambient);

      // 描画グループ（深度・責務分離）。entityGroup が深度ソート対象。
      this.worldGroup = new THREE.Group();
      this.worldGroup.name = 'world';
      this.entityGroup = new THREE.Group();
      this.entityGroup.name = 'entity';
      this.fxGroup = new THREE.Group();
      this.fxGroup.name = 'fx';
      this.shadowGroup = new THREE.Group();
      this.shadowGroup.name = 'shadow';
      this.scene.add(this.worldGroup, this.shadowGroup, this.entityGroup, this.fxGroup);
    }

    render(camera: THREE.Camera): void {
      this.renderer.render(this.scene, camera);
    }

    /**
     * 内部解像度は 1280×720 固定のまま、canvas の CSS 表示サイズを
     * アスペクト比を保ったまま与えられた領域に収める（レターボックス）。
     */
    resize(cssWidth: number, cssHeight: number): void {
      const targetAspect = INTERNAL_WIDTH / INTERNAL_HEIGHT;
      const containerAspect = cssWidth / cssHeight;
      let displayWidth = cssWidth;
      let displayHeight = cssHeight;
      if (containerAspect > targetAspect) {
        // 横が余る → 高さ基準
        displayWidth = Math.round(cssHeight * targetAspect);
      } else {
        // 縦が余る → 幅基準
        displayHeight = Math.round(cssWidth / targetAspect);
      }
      const canvas = this.renderer.domElement;
      canvas.style.width = `${displayWidth}px`;
      canvas.style.height = `${displayHeight}px`;
      // 内部解像度（drawingBufferサイズ）は変えない＝滲み防止。
      this.renderer.setSize(INTERNAL_WIDTH, INTERNAL_HEIGHT, false);
    }

    dispose(): void {
      this.renderer.dispose();
    }
  }
  ```
