  color: number;
  pixelW?: number;
  pixelH?: number;

function toCss(color: number): string {
  return '#' + color.toString(16).padStart(6, '0');
}

export function createPlaceholderTexture(opts: PlaceholderTextureOptions): THREE.Texture {
  const w = opts.pixelW ?? 64;
  const h = opts.pixelH ?? 96;
  const showMark = opts.showFacingMark ?? true;

  const canvas = document.createElement('canvas');
  canvas.width = w;
  canvas.height = h;
  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('placeholder: 2d context unavailable');

  // 透明背景（アルファ抜きビルボードを模す）
  ctx.clearRect(0, 0, w, h);

  // 本体（角丸矩形＝立ち姿のシルエット代用）
  const bodyMargin = Math.round(w * 0.18);
  const bodyTop = Math.round(h * 0.08);
  const bodyBottom = Math.round(h * 0.92);
  ctx.fillStyle = toCss(opts.color);
  ctx.fillRect(bodyMargin, bodyTop, w - bodyMargin * 2, bodyBottom - bodyTop);

  // 輪郭（縁を締めて滲み確認しやすく）
  ctx.strokeStyle = 'rgba(16,19,28,0.85)';
  ctx.lineWidth = 2;
  ctx.strokeRect(bodyMargin, bodyTop, w - bodyMargin * 2, bodyBottom - bodyTop);

  // 上端＝頭側の白帯（上下が分かるように）
  ctx.fillStyle = 'rgba(244,244,245,0.9)';
  ctx.fillRect(bodyMargin, bodyTop, w - bodyMargin * 2, Math.round(h * 0.08));

  // 下端中央＝足元アンカーを示す三角マーカー（facing の見え方確認用）
  if (showMark) {
    const cx = w / 2;
    const baseY = h - 4;
    const tipY = h - Math.round(h * 0.16);
    ctx.fillStyle = 'rgba(59,130,246,0.95)';
    ctx.beginPath();
    ctx.moveTo(cx, tipY);
    ctx.lineTo(cx - 8, baseY);
    ctx.lineTo(cx + 8, baseY);
    ctx.closePath();
    ctx.fill();
  }

  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.magFilter = THREE.NearestFilter; // 非整数縮小の滲み回避（08 R1）
  texture.minFilter = THREE.NearestFilter;
  texture.generateMipmaps = false;
  texture.needsUpdate = true;
  return texture;

```typescript
import * as THREE from 'three';

export interface DepthSortable {
  readonly id: number;
  footScreenDepth(camera: THREE.Camera): number;
  readonly mesh: { renderOrder: number };
}

// 再利用バッファ（毎フレーム呼ばれるためアロケーション抑制）
const _v = new THREE.Vector3();

// 足元ワールド座標をカメラのビュー空間へ変換し、-z（手前ほど大きい）を深度として返す純関数。
export function footScreenDepth(camera: THREE.Camera, x: number, z: number, y = 0): number {
  _v.set(x, y, z);
  _v.applyMatrix4(camera.matrixWorldInverse);
  // ビュー空間では -z 方向がカメラ前方。手前ほど z が大きい(0に近い負値)→ -z が大きい。
  return -_v.z;
}

// 奥→手前（深度昇順）に並べ替え renderOrder を 0..n-1 連番付与。第二キー=id。新配列を返す。
export function sortByDepth<T extends DepthSortable>(children: readonly T[], camera: THREE.Camera): T[] {
  const decorated = children.map((c) => ({ c, depth: c.footScreenDepth(camera) }));
  decorated.sort((a, b) => {
    if (a.depth !== b.depth) return a.depth - b.depth; // 昇順=奥が先
    return a.c.id - b.c.id;                            // 同深度はid昇順で安定化
  });
  const out: T[] = [];
  for (let i = 0; i < decorated.length; i++) {
    decorated[i].c.mesh.renderOrder = i;
    out.push(decorated[i].c);
  }
  return out;
}

import * as THREE from 'three';
import type { Facing } from '../types';
import { footScreenDepth } from './depth-sort';

export interface BillboardOptions {
  texture: THREE.Texture;
  worldSize: { w: number; h: number };
  blend?: 'normal' | 'add';
  framesPerRow?: number; // アトラス1行あたりコマ数（既定1=単一コマ）
}

let _nextId = 1;

export class BillboardSprite {
  readonly id: number;
  readonly mesh: THREE.Mesh;
  private readonly material: THREE.MeshBasicMaterial;
  private readonly texture: THREE.Texture;
  private readonly framesPerRow: number;
  private readonly worldW: number;
  private facingFlip = 1; // +1=左/その他, -1=右反転

  constructor(opts: BillboardOptions) {
    this.id = _nextId++;
    this.texture = opts.texture;
    this.framesPerRow = opts.framesPerRow ?? 1;
    this.worldW = opts.worldSize.w;

    // 足元アンカー: 下端中央が原点になるよう、高さの半分だけ上へ平行移動したジオメトリを作る。
    const geo = new THREE.PlaneGeometry(opts.worldSize.w, opts.worldSize.h);
    geo.translate(0, opts.worldSize.h / 2, 0); // 下端 y=0、上端 y=h

    this.material = new THREE.MeshBasicMaterial({
      map: opts.texture,
      transparent: true,
      alphaTest: 0.5,        // 縁抜け対策（08 §A1-2/R1）
      depthTest: false,      // 半透明混在のため明示ソート（08 §A1-4 B案）
      depthWrite: false,
      side: THREE.DoubleSide,
      blending: opts.blend === 'add' ? THREE.AdditiveBlending : THREE.NormalBlending,
    });

    this.mesh = new THREE.Mesh(geo, this.material);
    // 固定回転: カメラ仰角pitch30ぶん手前に倒す（板は地面に対して立て、毎フレーム lookAt しない）。
    this.mesh.rotation.set(THREE.MathUtils.degToRad(-30), 0, 0);
    this.mesh.frustumCulled = false; // 足元アンカーで境界が読みにくいためカリング無効
    this.setFrame(0);
  }

  setPosition(x: number, z: number, y = 0): void {
    this.mesh.position.set(x, y, z);
  }

  setFacing(facing: Facing): void {
    // right は left の水平反転（scale.x 反転）。down/left/up は正方向。
    this.facingFlip = facing === 'right' ? -1 : 1;
    this.mesh.scale.x = this.facingFlip;
  }

  setFrame(frame: number): void {
    const col = ((frame % this.framesPerRow) + this.framesPerRow) % this.framesPerRow;
    this.texture.repeat.set(1 / this.framesPerRow, 1);
    this.texture.offset.x = col / this.framesPerRow;
    this.texture.needsUpdate = true;
  }

  footScreenDepth(camera: THREE.Camera): number {
    return footScreenDepth(camera, this.mesh.position.x, this.mesh.position.z, this.mesh.position.y);
  }

  dispose(): void {
    this.mesh.geometry.dispose();
    this.material.dispose();
  }
}
```
