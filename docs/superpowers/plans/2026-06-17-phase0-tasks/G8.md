  ```typescript
  // 命名規則: <category>_<id>_<variant>_<dir>_<action>_<frame>@<scale>.png
  // Phase0 は単色/簡易ビルボードを engine/placeholder.ts でコード生成する。

  export interface PlaceholderDef {
    id: string;            // pc_nagisa / sk_gen / en_onibidoji
    color: number;         // 単色プレースホルダー色（0xRRGGBB）
    worldSize: { w: number; h: number };

  export const PLACEHOLDERS: Record<'pc_nagisa' | 'sk_gen' | 'en_onibidoji', PlaceholderDef> = {
    pc_nagisa:   { id: 'pc_nagisa',   color: 0x6fa8dc, worldSize: { w: 1,   h: 1.8 } }, // 凪沙（藍）
    sk_gen:      { id: 'sk_gen',      color: 0x3c6e47, worldSize: { w: 1,   h: 1.6 } }, // 玄（烏天狗・木）

  };

  // fal.ai 差し替え時のスプライトファイル名を一意生成する純関数。
  export function assetFileName(parts: {
    category: 'pc' | 'sk' | 'en';
    id: string;
    variant: string;
    dir: 'down' | 'left' | 'right' | 'up';
    action: string;
    frame: number;
    scale: 1 | 2;
  }): string {
    const { category, id, variant, dir, action, frame, scale } = parts;
    return `${category}_${id}_${variant}_${dir}_${action}_${frame}@${scale}.png`;
  }
  ```

  ```typescript
  // module: src/ui/hud.ts | HP/気/穢バー HUD（デウス UIPanel 思想・固定枠・スクロールバー厳禁）
  // 内部解像度 1280×720 のステージ左下に固定枠を置く。値変化はバー幅(%)のみで表現する。

  const KEGARE_LIGHT = 40;
  const KEGARE_HEAVY = 70;

  function bar(label: string, fillColor: string): { root: HTMLDivElement; fill: HTMLDivElement; text: HTMLDivElement } {
    const root = document.createElement('div');
    root.style.cssText =
      'position:relative;width:240px;height:18px;margin-top:6px;' +
      'background:rgba(16,19,28,0.78);border:1px solid rgba(180,200,230,0.35);' +
      'border-radius:3px;overflow:hidden;';
    const fill = document.createElement('div');
    fill.style.cssText =
      `position:absolute;left:0;top:0;height:100%;width:100%;background:${fillColor};` +
      'transition:width 120ms linear;';
    const text = document.createElement('div');
    text.style.cssText =
      'position:absolute;left:8px;top:0;height:100%;display:flex;align-items:center;' +
      "font:600 11px/1 'Noto Sans JP',system-ui,sans-serif;color:#f4f4f5;" +
      'text-shadow:0 1px 2px rgba(0,0,0,0.9);letter-spacing:0.04em;pointer-events:none;';
    text.textContent = label;
    root.appendChild(fill);
    root.appendChild(text);
    return { root, fill, text };
  }

  export class Hud {
    private readonly panel: HTMLDivElement;
    private readonly hpBar: ReturnType<typeof bar>;
    private readonly kiBar: ReturnType<typeof bar>;
    private readonly kegareBar: ReturnType<typeof bar>;

    constructor(mount: HTMLElement) {
      this.panel = document.createElement('div');
      this.panel.className = 'hud-panel';
      // 固定枠（UIPanel）。stage(1280×720)基準の左下。スクロールバー厳禁=overflow hidden。
      this.panel.style.cssText =
        'position:absolute;left:24px;bottom:24px;width:268px;' +
        'padding:10px 12px 12px;box-sizing:border-box;' +
        'background:rgba(10,12,18,0.55);border:1px solid rgba(150,170,210,0.28);' +
        'border-radius:6px;overflow:hidden;pointer-events:none;z-index:10;';

      this.hpBar = bar('HP', 'linear-gradient(90deg,#9a2b2b,#e05555)');
      this.kiBar = bar('気', 'linear-gradient(90deg,#1f5f8b,#4aa3d6)');
      this.kegareBar = bar('穢', 'linear-gradient(90deg,#5a4a7a,#8a6ad0)');

      this.panel.appendChild(this.hpBar.root);
      this.panel.appendChild(this.kiBar.root);
      this.panel.appendChild(this.kegareBar.root);
      mount.appendChild(this.panel);
    }

    private static pct(cur: number, max: number): number {
      if (max <= 0) return 0;
      return Math.max(0, Math.min(100, (cur / max) * 100));
    }

    setHp(cur: number, max: number): void {
      const p = Hud.pct(cur, max);
      this.hpBar.fill.style.width = `${p}%`;
      this.hpBar.text.textContent = `HP ${Math.max(0, Math.floor(cur))}/${Math.floor(max)}`;
      this.hpBar.root.dataset.pct = p.toFixed(1);
    }

    setKi(cur: number, max: number): void {
      const p = Hud.pct(cur, max);
      this.kiBar.fill.style.width = `${p}%`;
      this.kiBar.text.textContent = `気 ${Math.max(0, Math.floor(cur))}/${Math.floor(max)}`;
      this.kiBar.root.dataset.pct = p.toFixed(1);
    }

    setKegare(value: number): void {
      const v = Math.max(0, Math.min(100, value));
      this.kegareBar.fill.style.width = `${v}%`;
      // 段階で色味を変える（平常=紫薄 / 軽い穢 / 重い穢=赤紫）。閾値は契約 40/70。
      let g = 'linear-gradient(90deg,#5a4a7a,#8a6ad0)';
      if (v >= KEGARE_HEAVY) g = 'linear-gradient(90deg,#6a1f5a,#c04a8a)';
      else if (v >= KEGARE_LIGHT) g = 'linear-gradient(90deg,#5a3a7a,#a05ad0)';
      this.kegareBar.fill.style.background = g;
      this.kegareBar.text.textContent = `穢 ${Math.floor(v)}`;
      this.kegareBar.root.dataset.value = v.toFixed(1);
    }

    dispose(): void {
      this.panel.remove();
    }
  }
  ```

  ```typescript
      this.panel.appendChild(this.label);
      this.panel.appendChild(this.gauge);
      mount.appendChild(this.panel);
    }

    show(): void {
      if (this.visible) return;
      this.visible = true;
      this.panel.style.display = 'block';

      this.panel.style.display = 'none';
      this.fill.style.width = '0%';

      const v = Math.max(0, Math.min(1, p)) * 100;
      this.fill.style.width = `${v}%`;
      this.fill.dataset.progress = (v / 100).toFixed(3);
    }
  ```

  ```typescript
    interface Window {
      __harness?: {
        ready: boolean;

      };

      const player = game.world?.manager.player;
      const enemies = game.world?.manager.entities.filter((e) => 'kind' in e) ?? [];
      switch (name) {
  ```

  ```javascript
  // module: tools/screenshot.mjs | Playwright headless Chrome 検証ハーネス
  // 使い方: node tools/screenshot.mjs [--shot=default|moved|battle|chinkon|all] [--out=shots/x.png] [--url=http://localhost:5173]
  // dev サーバ(npm run dev)を別ターミナルで起動しておくか、--spawn で自動起動する。
  import { chromium } from 'playwright';
  import { spawn } from 'node:child_process';
  import { mkdir } from 'node:fs/promises';
  import { setTimeout as sleep } from 'node:timers/promises';

  function arg(name, def) {
    const hit = process.argv.find((a) => a.startsWith(`--${name}=`));
    return hit ? hit.slice(name.length + 3) : def;
  }
  const hasFlag = (name) => process.argv.includes(`--${name}`);

  const SHOT = arg('shot', 'default');
  const OUT = arg('out', `shots/${SHOT}.png`);
  const URL = arg('url', 'http://localhost:5173');
  const SPAWN = hasFlag('spawn');
  const VIEWPORT = { width: 1280, height: 720 };
  const SCENARIOS = SHOT === 'all'
    ? ['default', 'moved', 'battle', 'chinkon']
    : [SHOT];

  async function waitForServer(url, tries = 60) {
    for (let i = 0; i < tries; i++) {
      try {
        const res = await fetch(url);
        if (res.ok) return true;
      } catch { /* not up yet */ }
      await sleep(500);
    }
    throw new Error(`dev server not reachable at ${url}`);
  }

  async function main() {
    let server = null;
    if (SPAWN) {
      server = spawn('npm', ['run', 'dev'], { stdio: 'ignore', detached: false });
    }
    await waitForServer(URL);
    await mkdir('shots', { recursive: true });

    const browser = await chromium.launch({ args: ['--use-gl=angle', '--enable-webgl'] });
    const page = await browser.newPage({ viewport: VIEWPORT, deviceScaleFactor: 1 });
    const errors = [];
    page.on('pageerror', (e) => errors.push(String(e)));
    page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text()); });

    await page.goto(URL, { waitUntil: 'networkidle' });
    await page.waitForFunction(() => window.__harness?.ready === true, { timeout: 15000 });

    for (const sc of SCENARIOS) {
      await page.evaluate((name) => window.__harness.scenario(name), sc);
      await sleep(600); // 数フレーム進める（固定タイムステップで状態反映）
      const out = SHOT === 'all' ? `shots/${sc}.png` : OUT;
      await page.screenshot({ path: out });
      console.log(`[screenshot] ${sc} -> ${out}`);
    }

    await browser.close();
    if (server) { process.kill(server.pid); }

    if (errors.length) {
      console.error(`[screenshot] page errors detected (${errors.length}):`);
      for (const e of errors) console.error('  ' + e);
      process.exit(1);
    }
    console.log('[screenshot] done, no page errors.');
  }

  main().catch((e) => { console.error(e); process.exit(1); });
  ```

  ```javascript
  // 純粋テスト→型チェック→本番ビルド→無エラー撮影 を順に走らせ、1つでも失敗したら非ゼロ終了。
  import { spawnSync } from 'node:child_process';

  const steps = [
    { name: 'vitest (純粋ロジック)', cmd: 'npx', args: ['vitest', 'run'] },
    { name: 'typecheck (tsc --noEmit)', cmd: 'npx', args: ['tsc', '--noEmit'] },

    { name: 'screenshot (headless smoke)', cmd: 'node', args: ['tools/screenshot.mjs', '--shot=all', '--spawn'] },
  ];

  let failed = 0;
  for (const s of steps) {
    console.log(`\n=== SMOKE: ${s.name} ===`);
    const r = spawnSync(s.cmd, s.args, { stdio: 'inherit' });
    if (r.status !== 0) {
      console.error(`[smoke] FAILED: ${s.name} (exit ${r.status})`);
      failed++;
    } else {
      console.log(`[smoke] OK: ${s.name}`);
    }
  }

  if (failed > 0) {
    console.error(`\n[smoke] ${failed} step(s) failed.`);
    process.exit(1);
  }
  console.log('\n[smoke] all steps passed.');
  ```
