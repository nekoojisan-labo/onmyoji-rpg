import type { Facing } from '../types';

export interface MoveVector { moveX: number; moveZ: number; }

// 生入力(各軸 -1/0/+1 など)を長さ≤1へ正規化。斜め同時押しでも速くならない
export function normalizeMove(rawX: number, rawZ: number): MoveVector {
  const len = Math.hypot(rawX, rawZ);
  if (len <= 1e-6) {
    return { moveX: 0, moveZ: 0 };
  }
  if (len <= 1) {
    return { moveX: rawX, moveZ: rawZ };
  }
  return { moveX: rawX / len, moveZ: rawZ / len };
}

// 移動ベクトルを4方向 Facing へ丸める。ゼロベクトルは fallback を維持
export function facingFromMove(moveX: number, moveZ: number, fallback: Facing): Facing {
  if (Math.abs(moveX) <= 1e-6 && Math.abs(moveZ) <= 1e-6) {
    return fallback;
  }
  if (Math.abs(moveX) >= Math.abs(moveZ)) {
    return moveX >= 0 ? 'right' : 'left';
  }
  return moveZ >= 0 ? 'down' : 'up';
}
```

import type { Vec2, Rect } from '../types';

// Rect は中心(x,z)・幅w・奥行h → 半幅 hw, 半奥行 hh
function halfW(rect: Rect): number { return rect.w / 2; }
function halfH(rect: Rect): number { return rect.h / 2; }

// 円(中心x,z・半径r)が矩形と重なるか(距離二乗判定)
export function circleIntersectsRect(x: number, z: number, r: number, rect: Rect): boolean {
  const hw = halfW(rect);
  const hh = halfH(rect);
  const nearestX = Math.max(rect.x - hw, Math.min(x, rect.x + hw));
  const nearestZ = Math.max(rect.z - hh, Math.min(z, rect.z + hh));
  const dx = x - nearestX;
  const dz = z - nearestZ;
  return dx * dx + dz * dz < r * r;
}

// 1矩形に対する最小押し戻し後の円中心。軸分離で押し出しの少ない方向のみ動かす=スライド
export function resolveCircleRect(x: number, z: number, r: number, rect: Rect): Vec2 {
  if (!circleIntersectsRect(x, z, r, rect)) {
    return { x, z };
  }
  const hw = halfW(rect);
  const hh = halfH(rect);
  const nearestX = Math.max(rect.x - hw, Math.min(x, rect.x + hw));
  const nearestZ = Math.max(rect.z - hh, Math.min(z, rect.z + hh));
  const dx = x - nearestX;
  const dz = z - nearestZ;
  const distSq = dx * dx + dz * dz;

  if (distSq > 1e-12) {
    // 角の外側(辺の外)で食い込み → 最近点から半径方向に押し出す
    const dist = Math.sqrt(distSq);
    const push = r - dist;
    return { x: x + (dx / dist) * push, z: z + (dz / dist) * push };
  }

  // 中心が矩形内部 → 各辺への貫入量を比較し最小軸で押し出す(スライド)
  const penLeft = x - (rect.x - hw) + r;   // 左辺へ抜ける距離
  const penRight = (rect.x + hw) - x + r;  // 右辺へ抜ける距離
  const penUp = z - (rect.z - hh) + r;     // z-側
  const penDown = (rect.z + hh) - z + r;   // z+側
  const minPen = Math.min(penLeft, penRight, penUp, penDown);
  if (minPen === penLeft) return { x: (rect.x - hw) - r, z };
  if (minPen === penRight) return { x: (rect.x + hw) + r, z };
  if (minPen === penUp) return { x, z: (rect.z - hh) - r };
  return { x, z: (rect.z + hh) + r };
}

// マップ境界(内側のみ可動)へ円中心をクランプ
export function clampToMapBounds(x: number, z: number, r: number, bounds: Rect): Vec2 {
  const hw = halfW(bounds);
  const hh = halfH(bounds);
  const minX = bounds.x - hw + r;
  const maxX = bounds.x + hw - r;
  const minZ = bounds.z - hh + r;
  const maxZ = bounds.z + hh - r;
  return {
    x: Math.max(minX, Math.min(x, maxX)),
    z: Math.max(minZ, Math.min(z, maxZ)),
  };
}

// ブラックリスト矩形群＋境界に対する総合衝突解決(UpdateContext.collide の実体)
export function resolveCollision(
  x: number, z: number, radius: number,
  blockers: readonly Rect[], bounds: Rect,
): Vec2 {
  let px = x;
  let pz = z;
  for (const rect of blockers) {
    const next = resolveCircleRect(px, pz, radius, rect);
    px = next.x;
    pz = next.z;
  }
  return clampToMapBounds(px, pz, radius, bounds);
}
```

```typescript
import { normalizeMove } from './move-vector';

export interface InputState {
  moveX: number; moveZ: number;
  fuda: boolean;
  ju1: boolean; ju2: boolean;
  dodge: boolean;
  summon: boolean;
  switchShiki: boolean;
  item: boolean;
  chinkon: boolean;
}

// エッジ系アクションの内部キー → InputState フィールド名
type EdgeKey = 'fuda' | 'ju1' | 'ju2' | 'dodge' | 'summon' | 'switchShiki' | 'item' | 'chinkon';

// 物理キー(KeyboardEvent.code) → アクション
const ACTION_BY_CODE: Record<string, EdgeKey> = {
  KeyJ: 'fuda',
  KeyK: 'ju1',
  KeyL: 'ju2',
  Space: 'dodge',
  Digit1: 'summon',
  KeyE: 'switchShiki',
  KeyQ: 'item',
  KeyR: 'chinkon', // 鎮魂キー = R
};

export class Input {
  private readonly target: HTMLElement | Window;
  private readonly held = new Set<string>();   // 現在押下中の code
  private readonly edges = new Set<EdgeKey>();  // 当フレームに発生したエッジ
  private readonly onKeyDown: (e: KeyboardEvent) => void;
  private readonly onKeyUp: (e: KeyboardEvent) => void;

  constructor(target: HTMLElement | Window = window) {
    this.target = target;
    this.onKeyDown = (e: KeyboardEvent) => {
      const code = e.code;
      // 移動・アクションキーはスクロール等の既定動作を抑止
      if (this.isGameCode(code)) e.preventDefault();
      if (this.held.has(code)) return; // リピート無視 → エッジは1回だけ
      this.held.add(code);
      const action = ACTION_BY_CODE[code];
      if (action) this.edges.add(action);
    };
    this.onKeyUp = (e: KeyboardEvent) => {
      this.held.delete(e.code);
    };
    (this.target as EventTarget).addEventListener('keydown', this.onKeyDown as EventListener);
    (this.target as EventTarget).addEventListener('keyup', this.onKeyUp as EventListener);
  }

  private isGameCode(code: string): boolean {
    if (ACTION_BY_CODE[code]) return true;
    return (
      code === 'KeyW' || code === 'KeyA' || code === 'KeyS' || code === 'KeyD' ||
      code === 'ArrowUp' || code === 'ArrowDown' || code === 'ArrowLeft' || code === 'ArrowRight'
    );
  }

  // 当フレームの状態を返し edge をリセット
  sample(): InputState {
    let rawX = 0;
    let rawZ = 0;
    if (this.held.has('KeyD') || this.held.has('ArrowRight')) rawX += 1;
    if (this.held.has('KeyA') || this.held.has('ArrowLeft')) rawX -= 1;
    if (this.held.has('KeyS') || this.held.has('ArrowDown')) rawZ += 1; // z+ = 画面手前
    if (this.held.has('KeyW') || this.held.has('ArrowUp')) rawZ -= 1;   // z- = 画面奥
    const { moveX, moveZ } = normalizeMove(rawX, rawZ);

    const state: InputState = {
      moveX, moveZ,
      fuda: this.edges.has('fuda'),
      ju1: this.edges.has('ju1'),
      ju2: this.edges.has('ju2'),
      dodge: this.edges.has('dodge'),
      summon: this.edges.has('summon'),
      switchShiki: this.edges.has('switchShiki'),
      item: this.edges.has('item'),
      chinkon: this.edges.has('chinkon'),
    };
    this.edges.clear();
    return state;
  }

  dispose(): void {
    (this.target as EventTarget).removeEventListener('keydown', this.onKeyDown as EventListener);
    (this.target as EventTarget).removeEventListener('keyup', this.onKeyUp as EventListener);
    this.held.clear();
    this.edges.clear();
  }
}
```

- [ ] `src/engine/camera.ts` を実装（完全なクラス。G2 の既存コンストラクタ/`resize` は下記へ統合。yaw45/pitch30 の固定オフセットは G2 の値をそのまま使用）:
```typescript
import * as THREE from 'three';
import type { Rect } from '../types';
import { CAMERA_DEADZONE_X, CAMERA_DEADZONE_Z, CAMERA_FOLLOW_LERP } from '../config';

// yaw45/pitch30 固定のカメラ方向ベクトル(ターゲットからカメラ位置への単位オフセット)
// 仰角30°→ y成分 = sin30 = 0.5、水平面成分 = cos30。yaw45 で水平面を等分。
const PITCH_RAD = (30 * Math.PI) / 180;
const YAW_RAD = (45 * Math.PI) / 180;
const OFFSET_DIST = 50; // ターゲットからカメラまでの距離(unit)。オルソなので見た目スケール非依存
const CAM_OFFSET = new THREE.Vector3(
  Math.cos(PITCH_RAD) * Math.sin(YAW_RAD),
  Math.sin(PITCH_RAD),
  Math.cos(PITCH_RAD) * Math.cos(YAW_RAD),
).multiplyScalar(OFFSET_DIST);

export class IsoCamera {
  readonly camera: THREE.OrthographicCamera;
  private viewWorldWidth: number;
  private targetCenter = new THREE.Vector3(0, 0, 0); // 追従後のワールド中点
  private snapOffset = new THREE.Vector3(0, 0, 0);    // ピクセルスナップ補正

  constructor(viewWorldWidth: number, aspect: number) {
    this.viewWorldWidth = viewWorldWidth;
    const halfW = viewWorldWidth / 2;
    const halfH = halfW / aspect;
    this.camera = new THREE.OrthographicCamera(-halfW, halfW, halfH, -halfH, -1000, 1000);
    this.applyCenter();
  }

  // 中点(targetCenter)を基準にカメラ位置と注視点を更新
  private applyCenter(): void {
    const c = this.targetCenter;
    this.camera.position.set(
      c.x + CAM_OFFSET.x + this.snapOffset.x,
      c.y + CAM_OFFSET.y,
      c.z + CAM_OFFSET.z + this.snapOffset.z,
    );
    this.camera.up.set(0, 1, 0);
    this.camera.lookAt(c.x + this.snapOffset.x, c.y, c.z + this.snapOffset.z);
    this.camera.updateMatrixWorld();
  }

  // デッドゾーン付きlerp追従。ターゲットが矩形デッドゾーンを出た分だけ寄せる
  follow(targetX: number, targetZ: number): void {
    const dx = targetX - this.targetCenter.x;
    const dz = targetZ - this.targetCenter.z;
    let desiredX = this.targetCenter.x;
    let desiredZ = this.targetCenter.z;
    if (Math.abs(dx) > CAMERA_DEADZONE_X) {
      desiredX = targetX - Math.sign(dx) * CAMERA_DEADZONE_X;
    }
    if (Math.abs(dz) > CAMERA_DEADZONE_Z) {
      desiredZ = targetZ - Math.sign(dz) * CAMERA_DEADZONE_Z;
    }
    this.targetCenter.x += (desiredX - this.targetCenter.x) * CAMERA_FOLLOW_LERP;
    this.targetCenter.z += (desiredZ - this.targetCenter.z) * CAMERA_FOLLOW_LERP;
    this.snapOffset.set(0, 0, 0); // follow 後はスナップ補正を一旦解除(後段で再計算)
    this.applyCenter();
  }

  // マップ境界へ中点をクランプ。可視半幅ぶん内側に留め、外側の空白が見えないようにする
  clampToBounds(bounds: Rect): void {
    const halfViewW = this.viewWorldWidth / 2;
    const aspect = (this.camera.right - this.camera.left) / (this.camera.top - this.camera.bottom);
    const halfViewH = halfViewW / aspect;
    const hw = bounds.w / 2;
    const hh = bounds.h / 2;
    const minX = bounds.x - hw + halfViewW;
    const maxX = bounds.x + hw - halfViewW;
    const minZ = bounds.z - hh + halfViewH;
    const maxZ = bounds.z + hh - halfViewH;
    // マップが視界より小さい場合は中央へ固定
    this.targetCenter.x = minX > maxX ? bounds.x : Math.max(minX, Math.min(this.targetCenter.x, maxX));
    this.targetCenter.z = minZ > maxZ ? bounds.z : Math.max(minZ, Math.min(this.targetCenter.z, maxZ));
    this.applyCenter();
  }

  // 描画前ピクセルスナップ。中点を内部解像度ピクセル格子へ丸めサブピクセル滲みを防ぐ
  applyPixelSnap(internalWidth: number, internalHeight: number): void {
    const aspect = (this.camera.right - this.camera.left) / (this.camera.top - this.camera.bottom);
    const halfViewH = (this.viewWorldWidth / 2) / aspect;
    const unitsPerPixelX = this.viewWorldWidth / internalWidth;
    const unitsPerPixelZ = (halfViewH * 2) / internalHeight;
    const snappedX = Math.round(this.targetCenter.x / unitsPerPixelX) * unitsPerPixelX;
    const snappedZ = Math.round(this.targetCenter.z / unitsPerPixelZ) * unitsPerPixelZ;
    this.snapOffset.set(snappedX - this.targetCenter.x, 0, snappedZ - this.targetCenter.z);
    this.applyCenter();
  }

  resize(aspect: number): void {
    const halfW = this.viewWorldWidth / 2;
    const halfH = halfW / aspect;
    this.camera.left = -halfW;
    this.camera.right = halfW;
    this.camera.top = halfH;
    this.camera.bottom = -halfH;
    this.camera.updateProjectionMatrix();
    this.applyCenter();
  }
}
```
